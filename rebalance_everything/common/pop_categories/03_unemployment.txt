ruler_unemployment = {
	rank = 4
	clothes_texture_index = 1

	display_category = ruler

	demotion_time = 360

	pop_group_modifier = {
		pop_housing_usage_base = 1
		pop_amenities_usage_base = 1
		pop_happiness = -0.60
	}

	# scope: pop to be resettled
	allow_resettlement = {
		always = yes
	}

	resettlement_cost = {
		energy = 500
	}

	resettlement_cost = {
		trigger = {
			owner = {
				NOT = { has_valid_civic = civic_corvee_system }
			}
		}
		unity = 50
	}

	resources = {
		category = pop_category_rulers
	}
}
