triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = { is_regular_empire = yes }
		NOR = {
			has_building = building_physics_lab
			has_building = building_society_lab
			has_building = building_engineering_lab
		}
	}
	job_physicist_add = $AMOUNT$
	job_biologist_add = $AMOUNT$
	job_engineer_add = $AMOUNT$
}

	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
		}
		is_planet_class = pc_city

		NOR = {
			has_building = building_physics_lab

			has_building = building_engineering_lab
		}
	}
	job_physicist_add = $AMOUNT$
	job_biologist_add = $AMOUNT$
	job_engineer_add = $AMOUNT$
	mult = 2	# Ecus get 3× as many jobs
}
