triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
		}
	}
	job_bureaucrat_add = $AMOUNT$
}

triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
		}
		is_planet_class = pc_city
		NOT = {	has_modifier = penal_colony }
	}
	job_bureaucrat_add = $AMOUNT$
	mult = 2	# Ecus get 3× as many jobs
}
